```
cd D:\GitHub\ieumi-kiosk\ieumi-server
git pull                         # 최신 코드
# 서버는 창 없이 예약 작업 "Ieumi Kiosk"로 돈다 (2026-10-03부터, pc4 계정·로그인 불필요·부팅 시 자동)
#   재시작(코드 변경 시): node 프로세스만 끄면 boot-kiosk.bat 루프가 10초 뒤 새 코드로 다시 띄움
#     Get-NetTCPConnection -LocalPort 8791 -State Listen | % { Stop-Process -Id $_.OwningProcess -Force }
#   로그: ieumi-server\logs\boot-kiosk.log   ·   상태: Get-ScheduledTask 'Ieumi Kiosk'
#   start-kiosk.bat 을 따로 켜지 말 것 (포트 충돌로 둘 다 재시작 반복)
node db\qa-eval.js               # QA 전체 평가 (실측 ~$2.6, qa-runs\ 에 회차별 기록 + 직전 회차 비교)
node db\qa-eval.js --failed      # 구멍만 재테스트 (~$0.3, 회차로 기록 안 함). 배포 전엔 전체!
node db\qa-review-export.js      # 최근 회차 → 검수 페이지용 데이터 (Claude가 페이지에 올림, 크레딧 0)
node db\qa-report.js 3 "메모"    # 실무자용 리포트 HTML (크레딧 0)
node db\audit.js                 # 학습 상태 감사 (크레딧 0)
node db\sync-sources.js s30 --force   # 특정 서비스 재학습
node db\set-voice.js             # 목소리 vian 설정
```
